            "ตอนนี้เธอคิดว่าตัวเองกำลังพยายามสู้กับเรื่องอะไรอยู่หรือเปล่า?",

            "เพลงไหนหรือคำพูดไหนที่ช่วยฮีลใจเธอได้ดีที่สุดในช่วงนี้?",

            "เธอคิดว่าสังคมหรือโรงเรียนแบบไหนที่จะช่วยให้เธอเติบโตได้อย่างสบายใจที่สุด?",
            "นิยามคำว่า 'ความสำเร็จ' ของเธอในตอนนี้คืออะไร?",
            "หากเพื่อนข้างๆ กำลังรู้สึกเศร้า เธออยากบอกเขาคำว่าอะไรที่ธรรมดาแต่จริงใจที่สุด?"
        ];

        let availableQuestions = [...questions];

        const homeView = document.getElementById('home-view');
        const cardView = document.getElementById('card-view');
        const cardContainer = document.getElementById('card-container');
        const questionText = document.getElementById('question-text');

        // ฟังก์ชันเริ่มสุ่มการ์ดจากหน้าแรก
        function startDrawing() {
            // ซ่อนหน้าแรก แสดงหน้าการ์ด
            homeView.classList.add('hidden-view');
            cardView.classList.remove('hidden-view');
            
            // รีเซ็ตสถานะการ์ดให้คว่ำอยู่
            cardContainer.classList.remove('flip');
            
            // สุ่มคำถามมารอไว้ล่วงหน้า
            setRandomQuestion();

            // ให้การ์ดพลิกอัตโนมัติหลังจากเปิดหน้านี้สักพักสั้นๆ
            setTimeout(() => {
                cardContainer.classList.add('flip');
            }, 300);
        }

        // ฟังก์ชันสุ่มคำถามใหม่
        function drawNewCard() {
            // ถ้าการ์ดหงายอยู่ ให้คว่ำการ์ดลงก่อน
            if (cardContainer.classList.contains('flip')) {
                cardContainer.classList.remove('flip');
                
                // รอให้การ์ดคว่ำเสร็จ (0.6s) แล้วเปลี่ยนคำถาม จากนั้นพลิกกลับขึ้นมาใหม่
                setTimeout(() => {
                    setRandomQuestion();
                    cardContainer.classList.add('flip');
                }, 600);
            } else {
                // ถ้าการ์ดคว่ำอยู่ (กรณีบั๊กหรือผู้ใช้กดเร็ว) ก็เปลี่ยนคำถามและพลิกเลย
                setRandomQuestion();
                cardContainer.classList.add('flip');
            }
        }

        // ฟังก์ชันจัดการตรรกะการสุ่มคำถาม
        function setRandomQuestion() {
            // ถ้ารันคำถามครบแล้ว ให้รีเซ็ตใหม่
            if (availableQuestions.length === 0) {
                availableQuestions = [...questions];
            }

            // สุ่ม Index
            const randomIndex = Math.floor(Math.random() * availableQuestions.length);
            
            // ดึงคำถามออกมาและลบออกจาก array ชั่วคราวเพื่อไม่ให้ซ้ำติดๆ กัน
            const selectedQuestion = availableQuestions.splice(randomIndex, 1)[0];
            
            // ใส่ข้อความลงในการ์ด
            questionText.textContent = selectedQuestion;
        }

        // ฟังก์ชันกลับหน้าแรก
        function goHome() {
            // คว่ำการ์ดกลับ
            cardContainer.classList.remove('flip');
            
            // รอแอนิเมชันนิดหน่อยก่อนสลับหน้า
            setTimeout(() => {
                cardView.classList.add('hidden-view');
                homeView.classList.remove('hidden-view');
            }, 300);
        }
